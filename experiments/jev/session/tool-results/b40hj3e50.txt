// Run: node /tmp/jev-research/results/exp-main-container/build-candidates.mjs
import { createRequire } from 'node:module';
import { readFileSync, readdirSync, writeFileSync, statSync } from 'node:fs';
import { join, basename } from 'node:path';

const require = createRequire(import.meta.url);
const REPO = '/home/alexi/git/defuddle';
const { Defuddle } = require(join(REPO, 'dist/node.js'));

const MIN_WORDS = 30;
const MAX_CANDS = 12;
const TRUTH_COVERAGE = 0.9;

function normalize(text) {
	return (text || '').normalize('NFKC').replace(/[​‌‍﻿]/g, '').replace(/\s+/g, ' ').trim().toLowerCase();
}
function words(text) { return normalize(text).split(' ').filter(Boolean); }
function shingles(ws, n = 5) {
	const out = [];
	for (let i = 0; i + n <= ws.length; i++) out.push(ws.slice(i, i + n).join(' '));
	return out;
}
function className(el) {
	const c = typeof el.className === 'string' ? el.className : (el.getAttribute && el.getAttribute('class')) || '';
	return c.split(/[\t\n\f\r ]+/).filter(Boolean);
}
// Text of an element: skips script/style/noscript/template subtrees, inserts whitespace at block
// boundaries and <br> (linkedom textContent neither skips scripts nor separates blocks).
const BLOCK_TAGS = new Set([...require(join(REPO, 'dist/constants.js')).BLOCK_LEVEL_ELEMENTS, 'br', 'tr', 'td', 'th', 'li', 'option', 'img', 'hr', 'caption', 'label', 'button']);
const SKIP_TAGS = new Set(['script', 'style', 'noscript', 'template']);
function visibleText(el, opts = {}) {
	const parts = [];
	(function walk(node) {
		if (node.nodeType === 3) { parts.push(node.data); return; }
		if (node.nodeType !== 1) return;
		const tag = node.tagName.toLowerCase();
		if (SKIP_TAGS.has(tag)) return;
		if (opts.skipCode && (tag === 'pre' || tag === 'code')) { parts.push(' '); return; }
		const block = BLOCK_TAGS.has(tag);
		if (block) parts.push(' ');
		for (const child of node.childNodes) walk(child);
		if (block) parts.push(' ');
	})(el);
	return parts.join('');
}
function cleanPageHtml(html) {
	return html
		.replace(/<(script|style|noscript|template)\b[\s\S]*?<\/\1>/gi, '')
		.replace(/<svg\b[\s\S]*?<\/svg>/gi, '<svg></svg>')
		.replace(/<!--[\s\S]*?-->/g, '')
		.replace(/\s(src|href|srcset|data-src|data-srcset|poster|content)="data:[^"]{40,}"/gi, ' $1="data:..."')
		.replace(/="([^"]{150})[^"]*"/g, '="$1..."')
		.replace(/\s+/g, ' ')
		.replace(/> </g, '><');
}
// Order-insensitive containment: fraction of output words (multiset) present in the candidate's words.
function multisetCoverage(outWs, candWs) {
	const counts = new Map();
	for (const w of candWs) counts.set(w, (counts.get(w) || 0) + 1);
	let hit = 0;
	for (const w of outWs) { const c = counts.get(w) || 0; if (c > 0) { hit++; counts.set(w, c - 1); } }
	return outWs.length ? hit / outWs.length : 0;
}
const estimateTokens = (text, isHtml = false) => Math.ceil(text.length / (isHtml ? 3.5 : 4));

	}
	entry.extractorType = result.extractorType || null;
	entry.contentSelector = result.debug?.contentSelector || '';
	entry.title = result.title || '';
	if (result.extractorType) { entry.status = 'skipped_extractor'; log.push(entry); continue; }

	// Output text (what defuddle kept) -> shingles
	const outDoc = parseLinkedomHTML(`<html><body>${result.content}</body></html>`);
	const outWords = words(visibleText(outDoc.body));
	const outShingles = shingles(outWords);
	entry.outputWords = outWords.length;
	if (outShingles.length === 0) { entry.status = 'skipped_tiny_output'; log.push(entry); continue; }

		let matches = [];

		for (const el of matches) if (!seen.has(el)) seen.set(el, index);
	});
	let cands = [];
	for (const [el, selectorIndex] of seen) {
		const text = visibleText(el);
		const ws = words(text);
		if (ws.length < MIN_WORDS) continue;
		let defuddleScore = (ENTRY_POINT_ELEMENTS.length - selectorIndex) * 40;
		try { defuddleScore += ContentScorer.scoreElement(el); } catch { /* ignore */ }
		cands.push({ el, selectorIndex, selector: ENTRY_POINT_ELEMENTS[selectorIndex], text, ws, defuddleScore, docIndex: docIndex.get(el) ?? 1e9 });
	}
	entry.candidatesBeforeCap = cands.length;
	if (cands.length > MAX_CANDS) {
		cands.sort((a, b) => b.defuddleScore - a.defuddleScore);
		cands = cands.slice(0, MAX_CANDS);
		entry.capped = true;
	}
	cands.sort((a, b) => a.docIndex - b.docIndex); // present in document order
	if (cands.length < 2) { entry.status = 'skipped_lt2_candidates'; entry.candidates = cands.length; log.push(entry); continue; }

	// Ground truth: smallest candidate covering >= 90% of output shingles
	cands.forEach((c, i) => {
		c.id = `c${i + 1}`;
		const sset = new Set(shingles(c.ws));
		let hit = 0;
		const missing = [];
		for (const s of outShingles) { if (sset.has(s)) hit++; else if (missing.length < 5) missing.push(s); }
		c.coverage = hit / outShingles.length;
		c.missing = missing;
		c.coverageWords = multisetCoverage(outWords, c.ws);
		if (outShinglesNoCode.length >= 5) {
			const sset2 = new Set(shingles(words(visibleText(c.el, { skipCode: true }))));
			let hit2 = 0;
			for (const s of outShinglesNoCode) if (sset2.has(s)) hit2++;
			c.coverageNoCode = hit2 / outShinglesNoCode.length;
		} else c.coverageNoCode = null;
	});
	cands.forEach(c => {
		c.measures = [];
		if (c.coverage >= TRUTH_COVERAGE) c.measures.push('shingle');
		if (c.coverageNoCode !== null && c.coverageNoCode >= TRUTH_COVERAGE) c.measures.push('shingle_nocode');
		if (c.coverageWords >= TRUTH_COVERAGE) c.measures.push('words');
	});
	const covering = cands.filter(c => c.measures.length > 0).sort((a, b) => (a.ws.length - b.ws.length) || (a.el.outerHTML.length - b.el.outerHTML.length));
	const truth = covering.length ? covering[0].id : 'none_of_these';
	const truthMeasures = covering.length ? covering[0].measures : [];
	const truthStrictShingle = covering.length ? covering[0].measures.includes('shingle') : false;

	// Defuddle's pick mapped onto the candidate list
	let defuddleEl = null, defuddlePick = 'selector_not_found';
	try { defuddleEl = entry.contentSelector ? doc.querySelector(entry.contentSelector) : null; } catch { defuddleEl = null; }
	if (defuddleEl) {
		const match = cands.find(c => c.el === defuddleEl);
		if (match) defuddlePick = match.id;
		else {
			const containing = cands.filter(c => c.el.contains(defuddleEl)).sort((a, b) => (a.ws.length - b.ws.length) || (a.el.outerHTML.length - b.el.outerHTML.length));
			defuddlePick = containing.length ? `not_a_candidate(inside ${containing[0].id})` : 'not_a_candidate';
		}
	}

	// Candidate descriptions for the Jev state
	const described = cands.map(c => {
		const el = c.el;
		const text = normalize(c.text);
		let linkChars = 0;
		for (const a of Array.from(el.querySelectorAll('a'))) linkChars += normalize(visibleText(a)).length;
		const contains = cands.filter(o => o !== c && el.contains(o.el)).map(o => o.id);
		return {
			id: c.id,
			tag: el.tagName.toLowerCase(),
			id_attr: el.id || '',
			classes: className(el).slice(0, 8).join(' '),
			word_count: c.ws.length,
			paragraph_count: el.querySelectorAll('p').length,
			link_text_ratio: +(linkChars / Math.max(1, text.length)).toFixed(2),
			contains_candidates: contains,
			text_start: text.slice(0, 300),
			text_end: text.length > 450 ? text.slice(-150) : text.slice(300),
			// bookkeeping (not sent to Jev)
			_meta: { selector: c.selector, selectorIndex: c.selectorIndex, defuddleScore: Math.round(c.defuddleScore), coverage: +c.coverage.toFixed(3), coverageNoCode: c.coverageNoCode === null ? null : +c.coverageNoCode.toFixed(3), coverageWords: +c.coverageWords.toFixed(3), measures: c.measures, missing: c.missing, docIndex: c.docIndex }
		};
	});

	// Variant C: whole-page cleaned HTML with data-cand annotations
	cands.forEach(c => c.el.setAttribute('data-cand', c.id));
	const pageHtml = cleanPageHtml(doc.documentElement.outerHTML);
	const pageTokens = estimateTokens(pageHtml, true);

	kept.push({
		name, url, htmlBytes, title: entry.title, contentSelector: entry.contentSelector,
		outputWords: outWords.length, n_candidates: cands.length, candidatesBeforeCap: entry.candidatesBeforeCap, capped: !!entry.capped,
		truth, truthMeasures, truthStrictShingle, defuddlePick, candidates: described,
		pageHtml, pageTokens, pageEligibleC: pageTokens <= VARIANT_C_MAX_TOKENS,
	});
	entry.n_candidates = cands.length; entry.truth = truth; entry.truthMeasures = truthMeasures; entry.defuddlePick = defuddlePick; entry.pageTokens = pageTokens;
	log.push(entry);
}

// Phase 2 (live): ask Jev to pick the main article container. Variants:
//   A  = JSON candidate list (document order) + Choice + per-candidate Nouls
//   B  = same as A with candidate order reversed (state array and option order)
//   A2 = exact repeat of A (non-determinism baseline)
//   C  = whole-page cleaned HTML with data-cand annotations, Choice only
// Run: bash -ic 'node /tmp/jev-research/results/exp-main-container/run-jev.mjs --variant A [--limit N] [--names a,b] [--concurrency 6]'
import { readFileSync, writeFileSync, appendFileSync, existsSync, unlinkSync } from 'node:fs';
import { systemOne, mapConcurrent, usageTotals, estimateTokens } from '/tmp/jev-research/jev.mjs';

const args = Object.fromEntries(process.argv.slice(2).map((a, i, arr) => a.startsWith('--') ? [a.slice(2), arr[i + 1] && !arr[i + 1].startsWith('--') ? arr[i + 1] : true] : []).filter(Boolean));

const VERBOSE = !!args.verbose;

function publicCandidate(c) { const { _meta, ...rest } = c; return rest; }
function optionLabel(c) {
	let label = c.tag + (c.id_attr ? `#${c.id_attr}` : '') + (c.classes ? '.' + c.classes.split(' ').slice(0, 3).join('.') : '');
	label += `, ${c.word_count} words`;
	if (c.contains_candidates.length) label += `, contains ${c.contains_candidates.join(', ')}`;
	return label;
}
function buildListRequest(fx, reversed) {
	const cands = reversed ? [...fx.candidates].reverse() : fx.candidates;
	const state = { page_title: fx.title, url: fx.url, candidates: cands.map(publicCandidate) };
	const criteria = {};
	for (const c of cands) criteria[c.id] = optionLabel(c);
	criteria.none_of_these = 'No listed candidate is the smallest element containing the whole main article body';
	const questions = {
		main_container: {
			type: 'choice',
			instructions: `${QUESTION} The candidates are the entries of state.candidates; option names are their ids. Choose none_of_these if no candidate fits.`,
			criteria,
		},
	};
	for (const c of cands) {
		questions[`contains_all_${c.id}`] = { type: 'noul', instructions: `Does candidate ${c.id} contain the whole main article body (every paragraph, heading, figure, code block and footnote of the single main article)?` };
		questions[`non_article_${c.id}`] = { type: 'noul', instructions: `Does candidate ${c.id} contain substantial non-article material (navigation, sidebars, related links, comments, footers)?` };
	}
	return { state, questions };
}
function buildPageRequest(fx) {
	const criteria = {};
	for (const c of fx.candidates) criteria[c.id] = `the element whose opening tag carries data-cand="${c.id}" (<${c.tag}${c.id_attr ? ' id="' + c.id_attr + '"' : ''}>, ${c.word_count} words)`;
	criteria.none_of_these = 'No annotated element is the smallest element containing the whole main article body';
	return {
		state: fx.pageHtml,
		questions: {
			main_container: {
				type: 'choice',
				instructions: `${QUESTION} The state is the page HTML; the candidates are the elements annotated with a data-cand attribute. Choose none_of_these if no candidate fits.`,
				criteria,
			},
		},
	};
}

async function ask(req) {
	try {
		return { res: await systemOne({ model: MODEL, ...req }) };
	} catch (e) {
		return { error: { status: e.status ?? null, message: String(e.message || e).slice(0, 300) } };
	}
}

		// Split: Choice alone, then the Nouls (two requests, same state)
		record.split_on_400 = true;
		const q1 = { main_container: req.questions.main_container };
		const q2 = Object.fromEntries(Object.entries(req.questions).filter(([k]) => k !== 'main_container'));
		const r1 = await ask({ state: req.state, questions: q1 });
		const r2 = await ask({ state: req.state, questions: q2 });
		if (r1.res && r2.res) {
			res = { model: r1.res.model, answers: { ...r1.res.answers, ...r2.res.answers }, usage: { input_tokens: r1.res.usage.input_tokens + r2.res.usage.input_tokens, output_tokens: (r1.res.usage.output_tokens || 0) + (r2.res.usage.output_tokens || 0) }, latency_ms: r1.res.latency_ms + r2.res.latency_ms };
			error = null;
		} else error = r1.error || r2.error;
	}
	if (error) { record.error = error; }
	else {
		record.model = res.model;
		record.usage = res.usage;
		record.latency_ms = res.latency_ms;
		const mc = res.answers.main_container;
		record.choice = mc?.choice ?? null;
		record.confidence = mc?.confidence ?? null;
		record.probabilities = mc?.probabilities ?? null;

function relation(fx, pick, truth) {
	if (pick === truth) return 'correct';
	if (pick === 'none_of_these') return 'none_of_these';
	if (truth === 'none_of_these') return 'picked_when_label_none';
	const c = fx.candidates.find(x => x.id === pick);
	const t = fx.candidates.find(x => x.id === truth);
	if (!c || !t) return 'invalid';
	if (c.contains_candidates.includes(truth)) return 'ancestor_of_truth(over-inclusive)';
	if (t.contains_candidates.includes(pick)) return 'descendant_of_truth(under-inclusive)';
	return 'unrelated_branch';
}
function evalVariant(v, filter = () => true) {
	const recs = runs[v].filter(r => fixtures[r.name] && filter(fixtures[r.name], r));
	const ok = recs.filter(r => !r.error);
	const correct = ok.filter(r => r.choice === r.truth);
	const bands = {};
	for (const r of ok) { const b = band(r.confidence); bands[b] = bands[b] || { n: 0, correct: 0 }; bands[b].n++; if (r.choice === r.truth) bands[b].correct++; }
	for (const b of Object.keys(bands)) bands[b].accuracy_pct = pct(bands[b].correct, bands[b].n);
	const byN = {};
	for (const r of ok) { const k = r.n_candidates >= 4 ? '4+' : String(r.n_candidates); byN[k] = byN[k] || { n: 0, correct: 0 }; byN[k].n++; if (r.choice === r.truth) byN[k].correct++; }
	for (const k of Object.keys(byN)) byN[k].accuracy_pct = pct(byN[k].correct, byN[k].n);
	const rel = {};
	for (const r of ok) { const k = relation(fixtures[r.name], r.choice, r.truth); rel[k] = (rel[k] || 0) + 1; }
	const picks = {};
	for (const r of ok) { const fx = fixtures[r.name]; const c = fx.candidates.find(x => x.id === r.choice); const k = r.choice === 'none_of_these' ? 'none_of_these' : (c?.tag === 'body' ? 'body' : c?.tag || 'other'); picks[k] = (picks[k] || 0) + 1; }
	const lat = ok.map(r => r.latency_ms);
	return {
		n_fixtures: recs.length, n_errors: recs.length - ok.length, errors: recs.filter(r => r.error).map(r => ({ name: r.name, ...r.error })),
		correct: correct.length, accuracy_pct: pct(correct.length, ok.length),

}

function compare(v1, v2) {
	const pairs = runs[v1].filter(r => !r.error && R[v2][r.name] && !R[v2][r.name].error).map(r => [r, R[v2][r.name]]);
	const flips = pairs.filter(([a, b]) => a.choice !== b.choice);
	const dp = pairs.map(([a, b]) => Math.abs((a.probabilities?.[a.choice] ?? 0) - (b.probabilities?.[a.choice] ?? 0)));
	const dpAll = pairs.flatMap(([a, b]) => Object.keys(a.probabilities || {}).map(k => Math.abs((a.probabilities[k] ?? 0) - (b.probabilities?.[k] ?? 0))));
	return {
		n_pairs: pairs.length, choice_flips: flips.length, flip_rate_pct: pct(flips.length, pairs.length),
		flips: flips.map(([a, b]) => ({ name: a.name, truth: a.truth, [v1]: `${a.choice}@${a.confidence}`, [v2]: `${b.choice}@${b.confidence}` })),
		both_correct: pairs.filter(([a, b]) => a.choice === a.truth && b.choice === b.truth).length,
		either_correct: pairs.filter(([a, b]) => a.choice === a.truth || b.choice === b.truth).length,
		mean_abs_dp_chosen: +(dp.reduce((s, x) => s + x, 0) / Math.max(1, dp.length)).toFixed(3),
		max_abs_dp_any_option: +Math.max(...dpAll, 0).toFixed(2),
		mean_abs_dp_any_option: +(dpAll.reduce((s, x) => s + x, 0) / Math.max(1, dpAll.length)).toFixed(3),
	};
}

function noulPick(fx, nouls, thrContains, thrNon, useNon = true) {
	const ok = fx.candidates.filter(c => nouls[c.id] && nouls[c.id].contains_all != null && nouls[c.id].contains_all >= thrContains && (!useNon || nouls[c.id].non_article <= thrNon));

	ok.sort((a, b) => (a.word_count - b.word_count) || (a.contains_candidates.length - b.contains_candidates.length));

}
function evalNoul(v, thrContains, thrNon, useNon) {
